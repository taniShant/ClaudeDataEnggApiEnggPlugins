## Part 1: JWT Authentication and Authorization

### JWT Configuration

```java
package com.example.config;

@Configuration
@EnableWebSecurity
public class SecurityConfig {

    @Bean
    public SecurityFilterChain filterChain(HttpSecurity http) throws Exception {
        http
            .authorizeHttpRequests(authz -> authz
                .requestMatchers("/actuator/health").permitAll()
                .requestMatchers("/swagger-ui/**", "/v3/api-docs/**").permitAll()
                .requestMatchers("/public/**").permitAll()
                .anyRequest().authenticated()
            )
            .oauth2ResourceServer(oauth2 -> oauth2
                .jwt(jwt -> jwt
                    .decoder(jwtDecoder())
                    .jwtAuthenticationConverter(jwtAuthenticationConverter())
                )
            )
            .cors(Customizer.withDefaults())
            .csrf(csrf -> csrf.disable())  // JWT doesn't need CSRF
            .exceptionHandling(ex -> ex
                .authenticationEntryPoint(customAuthenticationEntryPoint())
                .accessDeniedHandler(customAccessDeniedHandler())
            )
            .httpBasic(Customizer.withDefaults());

        return http.build();
    }

    @Bean
    public JwtDecoder jwtDecoder() {
        return NimbusJwtDecoder.withPublicKey(rsaKey().publicKey()).build();
    }

    @Bean
    public JwtAuthenticationConverter jwtAuthenticationConverter() {
        JwtAuthenticationConverter converter = new JwtAuthenticationConverter();
        converter.setJwtGrantedAuthoritiesConverter(jwtGrantedAuthoritiesConverter());
        return converter;
    }

    private JwtGrantedAuthoritiesConverter jwtGrantedAuthoritiesConverter() {
        JwtGrantedAuthoritiesConverter converter = new JwtGrantedAuthoritiesConverter();
        converter.setAuthoritiesClaimName("roles");
        converter.setAuthorityPrefix("ROLE_");
        return converter;
    }

    @Bean
    public AuthenticationEntryPoint customAuthenticationEntryPoint() {
        return (request, response, authException) -> {
            response.setContentType("application/json");
            response.setStatus(HttpStatus.UNAUTHORIZED.value());
            response.getWriter().write("""
                {
                    "errorCode": "UNAUTHORIZED",
                    "message": "Authentication required",
                    "path": "%s"
                }
                """.formatted(request.getRequestURI()));
        };
    }

    @Bean
    public AccessDeniedHandler customAccessDeniedHandler() {
        return (request, response, accessDeniedException) -> {
            response.setContentType("application/json");
            response.setStatus(HttpStatus.FORBIDDEN.value());
            response.getWriter().write("""
                {
                    "errorCode": "FORBIDDEN",
                    "message": "Access denied",
                    "path": "%s"
                }
                """.formatted(request.getRequestURI()));
        };
    }

    @Bean
    public RSAKey rsaKey() {
        // Load from application.yml or environment variables
        return new RSAKey.Builder(rsaPublicKey()).build();
    }

    private RSAPublicKey rsaPublicKey() {
        // Implementation depends on key storage (PKCS#8, JWK Set, etc.)
        // For production: Load from secure key management service
        // For dev: Can load from classpath/environment
        return null;  // Placeholder
    }
}
```

### JWT Validation Aspect

```java
package com.example.aspect;

@Aspect
@Component
@Slf4j
public class JwtValidationAspect {

    @Before("@annotation(com.example.annotation.RequireRole)")
    public void validateRole(JoinPoint joinPoint) {
        MethodSignature signature = (MethodSignature) joinPoint.getSignature();
        RequireRole requireRole = signature.getMethod().getAnnotation(RequireRole.class);
        
        Authentication authentication = SecurityContextHolder.getContext().getAuthentication();
        if (authentication == null || !authentication.isAuthenticated()) {
            throw new AccessDeniedException("User not authenticated");
        }

        Set<String> requiredRoles = Set.of(requireRole.value());
        Set<String> userRoles = authentication.getAuthorities().stream()
            .map(GrantedAuthority::getAuthority)
            .collect(Collectors.toSet());

        if (!userRoles.stream().anyMatch(requiredRoles::contains)) {
            throw new AccessDeniedException(
                "User does not have required roles: " + String.join(", ", requiredRoles)
            );
        }
    }
}
```

## Part 2: CORS Configuration

### CORS Security Configuration

```java
package com.example.config;

@Configuration
public class CorsConfig {

    @Bean
    public WebMvcConfigurer corsConfigurer() {
        return new WebMvcConfigurer() {
            @Override
            public void addCorsMappings(CorsRegistry registry) {
                registry.addMapping("/api/**")
                    .allowedOrigins(getCorsAllowedOrigins())
                    .allowedMethods("GET", "POST", "PUT", "DELETE", "PATCH", "OPTIONS")
                    .allowedHeaders("*")
                    .exposedHeaders("Authorization", "X-Total-Count", "X-Total-Pages")
                    .allowCredentials(true)
                    .maxAge(3600);  // 1 hour

                // Public endpoints - more restrictive
                registry.addMapping("/public/**")
                    .allowedOrigins(getCorsAllowedOrigins())
                    .allowedMethods("GET", "POST", "OPTIONS")
                    .allowedHeaders("Content-Type")
                    .maxAge(1800);  // 30 minutes
            }
        };
    }

    private String[] getCorsAllowedOrigins() {
        String allowedOrigins = System.getenv("CORS_ALLOWED_ORIGINS");
        if (allowedOrigins == null || allowedOrigins.isEmpty()) {
            // Development default
            return new String[]{"http://localhost:3000", "http://localhost:8080"};
        }
        return allowedOrigins.split(",");
    }
}
```

## Part 3: Input Validation

### Comprehensive Input Validation

```java
package com.example.validation;

// Custom validators

@Target(ElementType.FIELD)
@Retention(RetentionPolicy.RUNTIME)
@Constraint(validatedBy = PhoneNumberValidator.class)
public @interface ValidPhoneNumber {
    String message() default "Invalid phone number format";
    Class<?>[] groups() default {};
    Class<? extends Payload>[] payload() default {};
}

public class PhoneNumberValidator implements ConstraintValidator<ValidPhoneNumber, String> {
    private static final Pattern PHONE_PATTERN = Pattern.compile("^\\+?[\\d\\s\\-()]{10,}$");

    @Override
    public boolean isValid(String value, ConstraintValidatorContext context) {
        if (value == null) return true;  // Handle @Nullable separately
        return PHONE_PATTERN.matcher(value).matches();
    }
}

@Target(ElementType.FIELD)
@Retention(RetentionPolicy.RUNTIME)
@Constraint(validatedBy = SafeSqlValidator.class)
public @interface SafeSql {
    String message() default "Input contains potentially dangerous SQL characters";
    Class<?>[] groups() default {};
    Class<? extends Payload>[] payload() default {};
}

public class SafeSqlValidator implements ConstraintValidator<SafeSql, String> {
    private static final Pattern SQL_INJECTION_PATTERN = 
        Pattern.compile("(?i)(DROP|DELETE|INSERT|UPDATE|UNION|SELECT|EXEC|EXECUTE)");

    @Override
    public boolean isValid(String value, ConstraintValidatorContext context) {
        if (value == null) return true;
        // Only basic check - actual SQL injection prevention via parameterized queries
        return !SQL_INJECTION_PATTERN.matcher(value).find();
    }
}
```

### Request Models (Java Records — no validation annotations)

DTOs MUST be Java Records. Per the project-wide rule, validation is performed by `RequestValidationFilter` against the OpenAPI spec — NOT per-DTO annotations.

```java
package com.example.model.request;

// ✅ CORRECT: Java Record — no Lombok, no validation annotations
public record CreateUserRequest(
    String name,
    String email,
    @Nullable String phone,   // spring-context @Nullable — NOT jakarta.validation
    String password,
    List<String> roles
) {}

public record SearchProductRequest(
    String query,
    Integer limit,      // default handled in service/controller layer, not DTO
    Integer offset
) {}
```

**Note:** Validation annotations (`@NotNull`, `@Email`, `@Size`, `@Pattern`, etc.) are FORBIDDEN on DTO records. Security constraints such as SQL-safe inputs and password strength are enforced by the upstream `RequestValidationFilter` which validates against the OpenAPI spec, not by DTO annotations. If additional runtime checks are needed, implement them in the service layer.

### Request Sanitization

```java
package com.example.util;

public class InputSanitizer {
    
    private static final Pattern HTML_PATTERN = Pattern.compile("<[^>]*>");
    private static final Pattern SCRIPT_PATTERN = Pattern.compile("<script[^>]*>.*?</script>", Pattern.CASE_INSENSITIVE | Pattern.DOTALL);

    public static String sanitizeHtml(String input) {
        if (input == null) return null;
        
        // Remove script tags
        String result = SCRIPT_PATTERN.matcher(input).replaceAll("");
        
        // Remove HTML tags
        result = HTML_PATTERN.matcher(result).replaceAll("");
        
        // HTML entity decode
        result = HtmlUtils.htmlUnescape(result);
        
        return result;
    }

    public static String sanitizeSql(String input) {
        if (input == null) return null;
        
        // Remove dangerous SQL keywords (basic check)
        return input.replaceAll("(?i)(DROP|DELETE|INSERT|UNION|EXEC)", "");
    }

    public static String sanitizeFilename(String filename) {
        if (filename == null) return null;
        
        // Remove path traversal characters
        filename = filename.replaceAll("\\.\\.[\\\\/]", "");
        
        // Remove special characters
        return filename.replaceAll("[^a-zA-Z0-9._-]", "");
    }
}
```

## Part 4: SQL Injection Prevention

### Parameterized Query Pattern

```java
package com.example.repository;

@Repository
public interface UserRepository extends JpaRepository<UserEntity, UUID> {
    
    // Using JPA named parameters (safe from SQL injection)
    Optional<UserEntity> findByEmail(String email);
    
    List<UserEntity> findByNameContainingIgnoreCase(String name);
    
    // Using @Query with named parameters
    @Query("SELECT u FROM UserEntity u WHERE u.name LIKE %:name% AND u.active = true")
    List<UserEntity> searchActiveUsers(@Param("name") String name);
    
    // Using @Query with ?1 (positional parameters) - also safe
    @Query("SELECT u FROM UserEntity u WHERE u.email = ?1 AND u.createdAt > ?2")
    List<UserEntity> findByEmailAfterDate(String email, LocalDateTime date);
}
```

### Repository Usage

```java
@Service
@RequiredArgsConstructor
public class UserServiceImpl implements UserService {
    private final UserRepository userRepository;

    // ✓ SAFE: Using JPA methods with parameters
    public Optional<User> findByEmail(String email) {
        return userRepository.findByEmail(email)
            .map(this::mapToUser);
    }

    // ✓ SAFE: Using @Query with named parameters
    public List<User> searchUsers(String name) {
        return userRepository.searchActiveUsers(name)
            .stream()
            .map(this::mapToUser)
            .collect(Collectors.toList());
    }

    // ✗ UNSAFE: String concatenation (DO NOT DO THIS)
    // public List<User> dangerousSearch(String query) {
    //     String sql = "SELECT * FROM users WHERE name LIKE '%" + query + "%'";
    //     // String concatenation allows SQL injection!
    // }

    // ✓ SAFE: Using RestClient with parameters
    public Payment getPaymentFromService(String paymentId) {
        return restClient.get()
            .uri("/payments/{id}", paymentId)  // Parameter binding
```

## Part 5: CSRF Protection

### CSRF Configuration

```java
package com.example.config;

@Configuration
public class CsrfConfig {

    @Bean
    public SecurityFilterChain csrfFilterChain(HttpSecurity http) throws Exception {
        http.csrf(csrf -> csrf
            .csrfTokenRepository(csrfTokenRepository())
            .csrfTokenRequestHandler(new CsrfTokenRequestAttributeHandler())
            .ignoringRequestMatchers(
                "/api/public/**",
                "/actuator/**"
            )
        );

        return http.build();
    }

    @Bean
    public CsrfTokenRepository csrfTokenRepository() {
        HttpSessionCsrfTokenRepository repository = new HttpSessionCsrfTokenRepository();
        repository.setHeaderName("X-CSRF-TOKEN");
        repository.setParameterName("_csrf");
        return repository;
    }
}
```

## Part 6: API Rate Limiting

### Rate Limiting Configuration

```java
package com.example.config;

@Configuration
public class RateLimitingConfig {

    @Bean
    public RateLimiter globalRateLimiter() {
        return RateLimiter.create(100);  // 100 requests per second
    }

    @Bean
    public Bucket bucket() {
        Bandwidth limit = Bandwidth.classic(100, Refill.intervally(100, Duration.ofSeconds(1)));
        return Bucket.builder()
            .addLimit(limit)
            .build();
    }
}

@Aspect
@Component
@Slf4j
public class RateLimitingAspect {
    
    private final Bucket bucket;

    public RateLimitingAspect(Bucket bucket) {
        this.bucket = bucket;
    }

    @Before("@annotation(com.example.annotation.RateLimit)")
    public void checkRateLimit(JoinPoint joinPoint) {
        if (!bucket.tryConsume(1)) {
            throw new RateLimitExceededException("Too many requests");
        }
    }
}

@ResponseStatus(HttpStatus.TOO_MANY_REQUESTS)
class RateLimitExceededException extends RuntimeException {
    public RateLimitExceededException(String message) {
        super(message);
    }
}
```

## Part 7: Secure Password Management

### Password Validation and Storage

```java
package com.example.util;

@Component
public class PasswordManager {
    
    private final PasswordEncoder passwordEncoder;

    public PasswordManager() {
        // Use BCrypt with strong cost factor
        this.passwordEncoder = new BCryptPasswordEncoder(12);
    }

    public String hashPassword(String rawPassword) {
        if (rawPassword == null || rawPassword.isEmpty()) {
            throw new IllegalArgumentException("Password cannot be empty");
        }
        return passwordEncoder.encode(rawPassword);
    }

    public boolean verifyPassword(String rawPassword, String hashedPassword) {
        return passwordEncoder.matches(rawPassword, hashedPassword);
    }

    public boolean isStrongPassword(String password) {
        if (password == null || password.length() < 8) return false;
        if (!password.matches(".*[a-z].*")) return false;      // lowercase
        if (!password.matches(".*[A-Z].*")) return false;      // uppercase
        if (!password.matches(".*\\d.*")) return false;        // digit
        if (!password.matches(".*[@$!%*?&].*")) return false;  // special char
        return true;
    }
}

public class PasswordResetValidator {
    
    private static final long TOKEN_EXPIRY_MINUTES = 15;

    public static String generateResetToken() {
        return UUID.randomUUID().toString();
    }

    public static boolean isTokenExpired(LocalDateTime createdAt) {
        return LocalDateTime.now().isAfter(
            createdAt.plusMinutes(TOKEN_EXPIRY_MINUTES)
        );
    }
}
```

## Part 8: Secure Headers

### Security Headers Configuration

```java
package com.example.config;

@Configuration
public class SecurityHeadersConfig {

    @Bean
    public SecurityFilterChain securityHeadersFilterChain(HttpSecurity http) throws Exception {
        http.headers(headers -> headers
            .contentSecurityPolicy(csp -> csp
                .policyDirectives("default-src 'self'; script-src 'self' trusted-cdn.example.com; style-src 'self' 'unsafe-inline'")
            )
            .xssProtection(xss -> xss
                .and()
                .headerValue("1; mode=block")
            )
            .frameOptions(frame -> frame.sameOrigin())
            .httpStrictTransportSecurity(hsts -> hsts
                .includeSubDomains(true)
                .maxAgeInSeconds(31536000)  // 1 year
            )
            .referrerPolicy(ref -> ref
                .policy(ReferrerPolicyHeaderWriter.ReferrerPolicy.STRICT_ORIGIN_WHEN_CROSS_ORIGIN)
            )
        );

        return http.build();
    }
}
```

### Response Header Filter

```java
@Component
public class SecurityHeadersFilter extends OncePerRequestFilter {

    @Override
    protected void doFilterInternal(HttpServletRequest request, 
            HttpServletResponse response, FilterChain filterChain)
            throws ServletException, IOException {
        
        // Prevent MIME type sniffing
        response.setHeader("X-Content-Type-Options", "nosniff");
        
        // Enable XSS protection
        response.setHeader("X-XSS-Protection", "1; mode=block");
        
        // Clickjacking protection
        response.setHeader("X-Frame-Options", "SAMEORIGIN");
        
        // HSTS
        response.setHeader("Strict-Transport-Security", 
            "max-age=31536000; includeSubDomains; preload");
        
        // CSP
        response.setHeader("Content-Security-Policy",
            "default-src 'self'; script-src 'self' trusted-cdn.example.com");
        
        filterChain.doFilter(request, response);
    }
}
```

### Internal API Initializer with Correlation ID

```java
@Configuration
public class InternalApiInitializerConfig {

    @Bean
    public Filter correlationIdFilter() {
        return new OncePerRequestFilter() {
            @Override
            protected void doFilterInternal(HttpServletRequest request, 
                    HttpServletResponse response, FilterChain filterChain)
                    throws ServletException, IOException {
                
                String correlationId = request.getHeader("X-Correlation-ID");
                if (correlationId == null) {
                    correlationId = UUID.randomUUID().toString();
                }
                
                MDC.put("correlationId", correlationId);
                response.setHeader("X-Correlation-ID", correlationId);
                
                try {
                    filterChain.doFilter(request, response);
                } finally {
                    MDC.clear();
                }
            }
        };
    }
}
```

```java
    }

    void testSqlInjectionPrevention() throws Exception {
        // Attempt SQL injection - should be safe

    }

        // Verify password is hashed
```
